# DISCONFIRMING PROBE for C18 — the orchestrator's, run before the brief.
# Classifies every .wat.bad by its ACTUAL failure kind on the current binary.
# ⛔ DO NOT TRUNCATE THE OUTPUT. A first version sliced stdout+stderr to 400 chars and
# reported 16 where the truth is 17: typed_if_match_bare_symbol_variant.wat.bad carries
# its MainSignature error at char 4441. Stable at 17 over three runs untruncated.
import subprocess, glob, collections
files = sorted(set(
    glob.glob('tests/**/*.wat.bad', recursive=True)
    + glob.glob('wat-scripts/**/*.wat.bad', recursive=True)
    + glob.glob('docs/**/*.wat.bad', recursive=True)))
kinds = collections.Counter(); mainfail = []
for f in files:
    p = subprocess.run(['./target/release/wat', f], capture_output=True, text=True, timeout=60)
    out = p.stdout + p.stderr          # NEVER slice this
    if 'MainSignature' in out:  k = 'MainSignatureError'; mainfail.append(f)
    elif 'CheckErrors' in out:  k = 'CheckErrors'
    elif p.returncode == 0:     k = '!! DID NOT FAIL'
    else:                       k = 'other'
    kinds[k] += 1
print(f"total .wat.bad driven: {len(files)}")
for k, v in kinds.most_common(): print(f"  {k:22} {v}")
for f in mainfail:
    h = open(f, errors='replace').read()
    tag = 'EXPIRED-PREMISE' if ('retired' in h.lower() or 'formerly' in h.lower()) else 'dies-before-wall'
    print(f"  {tag:16} {f}")
